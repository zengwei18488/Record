sudo systemctl stop usbguard 2>/dev/null || true
sudo systemctl disable usbguard 2>/dev/null || true
sudo systemctl mask usbguard 2>/dev/null || true

# 2. 同步停止與之連動的 D-Bus 接口
sudo systemctl stop usbguard-dbus 2>/dev/null || true

# 3. 清理 systemd 內的所有軟連結
sudo find /etc/systemd/ -name "*usbguard*" -delete 2>/dev/null || true
